model{

	pi<-3.141
	#c<-100
	
	
	isr0[1]~dbern(p[1])
	#p[1]~dbeta(mm,nn)
	logit(p[1])<-alp/(1-phi)
	
	mu[1]<-beta[1]/(1-beta[2])
	
	sig.r[1]<-sig/(1-beta[2]^2)
	Prec[1]<-1/(sig.r[1]^2)
	
	r[1]~dlnorm(mu[1],Prec[1])
	
	y[1]<-(isr0[1])+(1-isr0[1])*r[1]
	
	for(t in 2:T){
	
		isr0[t]~dbern(p[t])
		#p[t]~dbeta(mm,nn)	
		logit(p[t])=alp+phi*(p[t-1])
	
		mu[t]<-(1-isr0[t])*(beta[1]+beta[2]*log(r[t-1]))
		sig.r[t]<-(1-isr0[t]+0.001)*sig    #(isr0[t]+1/c)*sig
		Prec[t]<-1/(sig.r[t]^2)
		
		r[t]~dlnorm(mu[t],Prec[t])
		y[t]<-isr0[t]+(1-isr0[t])*r[t]
		}
		sig<-1/sqrt(tau2)
		tau2~dgamma(df1,df2)
		df1~dexp(0.1)
		df2~dexp(0.1)	
	
	
#	Priors:
    
	beta[1]~dnorm(0,tau2.beta)
	beta[2]~dunif(-0.99,0.99)
	
	alp~dnorm(0,.25)
	phi~dunif(-0.99,0.99)
	
	
	#mm~dexp(0.1)
	#nn~dexp(0.1)
	
	sig.beta<-1/sqrt(tau2.beta)
	tau2.beta~dgamma(nu1,nu2)
	
	nu1~dexp(0.1)
	nu2~dexp(0.1)	


# Predictive distribution:

	for(t in 1:T){
		rP[t]~dlnorm(mu[t],Prec[t])
		yP[t]<-isr0[t]+(1-isr0[t])*rP[t]
		}
}
